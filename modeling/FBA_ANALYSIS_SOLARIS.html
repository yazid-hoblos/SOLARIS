# Fonction to check whether a metabolite already exists in PCC7942 and creates one if it doesn't
def retrieve_or_create_metabolite(model_PCC7942_3HPP, met_id, name, compartment="c"):
    try:
        return model_PCC7942_3HPP.metabolites.get_by_id(met_id)
    except KeyError:
        m = Metabolite(id=met_id, name=name, compartment=compartment)
        model_PCC7942_3HPP.add_metabolites([m])
        return m

results = []
# Run FBA for each single gene deletion
for gene_id in gene_deletion_ids: 
    model_PCC7942_3HPP_individual_DEL = model_PCC7942_3HPP.copy()
    gene = model_PCC7942_3HPP_individual_DEL.genes.get_by_id(gene_id)
    gene.knock_out()
    solution = model_PCC7942_3HPP_individual_DEL.optimize()
    biomass_flux = solution.objective_value

    })
